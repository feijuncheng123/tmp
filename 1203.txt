from hdbscan import HDBSCAN
from hdbscan.validity import validity_index
from sklearn.metrics import silhouette_score
import numpy as np
import pandas as pd

def evaluate_hdbscan_params(X, 
                            min_cluster_sizes=[30,50,80,100,150,200],
                            min_samples=[5,10,15,20,None]):
    results = []
    
    for mcs M in min_cluster_sizes:
        for ms in min_samples:
            clusterer = HDBSCAN(min_cluster_size=mcs,
                                min_samples=ms if ms else None,  # None=自动等于mcs
                                cluster_selection_epsilon=0.0,
                                metric='euclidean')
            labels = clusterer.fit_predict(X)
            
            n_clusters = len(set(labels)) - (1 if -1 in labels else 0)
            noise_ratio = (labels == -1).mean()
            dbcv = validity_index(X.astype(np.float64), labels)  # 必须 float64
            
            # 1. 核心：DBCV（权重最高）
            score1 = dbcv                                        # 0~1
            
            # 2. 簇数量 + 噪声惩罚
            if 12 <= n_clusters <= 70 and noise_ratio < 0.35:
                score2 = 1.0
            elif n_clusters < 12:
                score2 = n_clusters / 12
            elif n_clusters > 70:
                score2 = 70 / n_clusters
            else:
                score2 = 1 - (noise_ratio - 0.3)/0.7
            
            # 3. 稳定性预估（简化版：用概率噪声点比例反向估计）
            stability_proxy = 1 - clusterer.probabilities_.mean() * 0.5
            
            # 综合分数（工业界真实权重）
            final_score = 0.55 * score1 + 0.30 * score2 + 0.15 * stability_proxy
            
            results.append({
                'min_cluster_size': mcs,
                'min_samples': ms,
                'n_clusters': n_clusters,
                'noise_ratio': noise_ratio,
                'dbcv': dbcv,
                'final_score': final_score,
                'labels': labels
            })
    
    df = pd.DataFrame(results).sort_values('final_score', ascending=False)
    best = df.iloc[0]
    print(f"最佳参数: min_cluster_size={best.min_cluster_size}, "
          f"min_samples={best.min_samples or best.min_cluster_size}")
    print(f"簇数: {best.n_clusters}, 噪声率: {best.noise_ratio:.1%}, DBCV: {best.dbcv:.3f}")
    
    return df, best.labels
