import numpy as np
import lightgbm as lgb
from scipy.special import digamma
from scipy.stats import t
from scipy.optimize import minimize

eps = 1e-9

def student_t_obj(preds, train_data):
    # preds is a 1D numpy array of length 3*n (concatenated: mu_block | logsigma_block | eta_block)
    # However, LightGBM will pass preds length == n by default; this concatenation trick
    # relies on LightGBM updating preds with the same length as labels and expects
    # custom objective to return arrays of that same length. Some users hack by passing
    # Dataset with label length = 3*n and duplicates labels; that is brittle.
    #
    # Here we implement the vectorized gradients given preds is concatenated as expected.
    y = train_data.get_label()
    n = y.shape[0]
    # Expect preds length = 3*n
    if preds.shape[0] != 3*n:
        raise ValueError("This objective expects preds length 3*n (mu|logsigma|eta). Got %d vs %d" % (preds.shape[0], 3*n))
    mu = preds[0:n]
    log_sigma = preds[n:2*n]
    eta = preds[2*n:3*n]  # eta = log(nu - 2)
    sigma = np.exp(log_sigma) + eps
    nu = 2.0 + np.exp(eta)  # ensure nu>2
    z = (y - mu) / sigma
    tval = 1.0 + (z*z) / nu
    B = (nu + 1.0) / nu

    # grads
    grad_mu = - B * z / (sigma * tval)         # dL/dmu
    grad_log_sigma = 1.0 - B * (z*z) / tval    # dL/d(log sigma)

    # dL/dnu (then chain to dL/deta)
    term1 = 0.5 / nu
    term2 = 0.5 * (digamma(nu/2.0) - digamma((nu+1.0)/2.0))
    term3 = 0.5 * np.log(np.maximum(tval, eps))
    term4 = - (nu + 1.0) * (z*z) / (2.0 * (nu*nu) * tval)
    dL_dnu = term1 + term2 + term3 + term4
    grad_eta = (nu - 2.0) * dL_dnu

    # hessians
    hess_mu = ( (nu + 1.0) / nu ) * (1.0 / (sigma*sigma)) * (1.0/tval - 2.0 * (z*z) / (nu * tval*tval))
    hess_log_sigma = 2.0 * (nu + 1.0) / nu * (z*z) * (1.0/tval - (z*z) / (nu * tval*tval))

    # for eta (nu param) we use a positive stable approximation for hessian
    hess_eta = np.maximum(1e-6, np.abs(grad_eta) + 1e-6)

    grad = np.concatenate([grad_mu, grad_log_sigma, grad_eta])
    hess = np.concatenate([hess_mu, hess_log_sigma, hess_eta])
    return grad, hess

# Example training wrapper (VERY experimental!)
def train_joint_model(X, y):
    n = X.shape[0]
    # Trick: we prepare a dataset with label repeated three times, and we will
    # also provide 'init_score' of length 3*n to seed preds blocks. This is hacky.
    # Many users instead choose to implement a custom boosting loop.
    y_dup = np.concatenate([y, y, y])
    dtrain = lgb.Dataset(X, label=y_dup)  # NOTE: this mismatches feature rows vs labels!
    # This is intentionally sketch — most LightGBM versions expect label length == n,
    # and using a label of length 3*n with X of n rows may error.
    # If you want to experiment, one approach is to tile X vertically 3 times too:
    X_trip = np.vstack([X, X, X])
    dtrain = lgb.Dataset(X_trip, label=y_dup)
    params = {'objective': 'regression', 'metric':'None', 'verbosity': -1}
    bst = lgb.train(params, dtrain, fobj=student_t_obj, num_boost_round=200)
    return bst




import numpy as np
import lightgbm as lgb
from scipy.special import gammaln

# =====================
# 1. Student-t logPDF
# =====================
def student_t_logpdf(y, mu, sigma, nu):
    """
    计算 Student-t 对数似然 (逐点)
    y: 实际值
    mu: 预测均值
    sigma: 尺度参数 > 0
    nu: 自由度参数 > 0
    """
    z = (y - mu) / sigma
    return (
        gammaln((nu + 1) / 2)
        - gammaln(nu / 2)
        - 0.5 * np.log(nu * np.pi)
        - np.log(sigma)
        - (nu + 1) / 2 * np.log(1 + z**2 / nu)
    )


# =====================
# 2. unpack预测向量
# =====================
def unpack_predictions(y_pred):
    """
    LightGBM 输出是一维向量，
    我们假设每个样本有 3 个参数 (mu, log_sigma, log_nu)。
    """
    y_pred = y_pred.reshape(-1, 3)
    mu = y_pred[:, 0]
    log_sigma = y_pred[:, 1]
    log_nu = y_pred[:, 2]
    sigma = np.exp(log_sigma)
    nu = np.exp(log_nu)
    return mu, sigma, nu, log_sigma, log_nu


# =====================
# 3. 自定义目标函数 (grad + hess)
# =====================
def student_t_nll_loss(y_pred, dataset):
    y_true = dataset.get_label()
    mu, sigma, nu, log_sigma, log_nu = unpack_predictions(y_pred)

    # 常用符号
    z = (y_true - mu) / sigma
    denom = 1 + (z**2) / nu

    # ------- grad -------
    # d/dmu
    grad_mu = (nu + 1) * z / (nu * denom * sigma)

    # d/dlog_sigma
    grad_log_sigma = -1 + (nu + 1) * (z**2) / (nu * denom)

    # d/dlog_nu (近似推导，确保数值稳定)
    grad_log_nu = (
        0.5 * (1 - (nu + 1) * (z**2) / (nu * denom))
    )

    grad = np.vstack([grad_mu, grad_log_sigma, grad_log_nu]).T.reshape(-1)

    # ------- hess -------
    # 为简化，这里返回一个近似的常数 hess（保证正定）
    hess = np.ones_like(grad) * 0.1

    return grad, hess


# =====================
# 4. 自定义评估函数 (NLL)
# =====================
def student_t_nll_eval(y_pred, dataset):
    y_true = dataset.get_label()
    mu, sigma, nu, _, _ = unpack_predictions(y_pred)
    nll = -student_t_logpdf(y_true, mu, sigma, nu).mean()
    return "student_t_nll", nll, False  # False = 越小越好

# student_t_two_stage.py
import numpy as np
import pandas as pd
import lightgbm as lgb
from scipy.stats import t
from scipy.optimize import minimize
from scipy.stats import median_abs_deviation

# ---------------------------
# 生成示例数据（你可替换为真实数据）
# ---------------------------
def gen_series(n=3000, seed=2025):
    rng = np.random.RandomState(seed)
    t_idx = np.arange(n)
    base = 1000 + 20*np.sin(2*np.pi*t_idx/288)
    noise = rng.normal(scale=3.0, size=n)
    spikes = (rng.rand(n) < 0.002).astype(float) * (np.abs(rng.standard_t(df=3, size=n)) * 60)
    y = base + noise + spikes
    return pd.Series(y)

# ---------------------------
# 简单特征构造（示例）
# ---------------------------
def make_features(series, L=288, stride=1):
    X_rows = []
    y = []
    idxs = []
    n = len(series)
    for start in range(0, n - L - 1, stride):
        win = series.iloc[start:start+L].values
        last = win[-1]
        mean_1h = win[-12:].mean()
        std_1h = win[-12:].std()
        mean_6h = win[-72:].mean()
        std_6h = win[-72:].std()
        slope = np.polyfit(np.arange(len(win[-24:])), win[-24:], 1)[0]
        X_rows.append([last, mean_1h, std_1h, mean_6h, std_6h, slope])
        y.append(series.iloc[start+L])  # predict next point
        idxs.append(series.index[start+L])
    X = pd.DataFrame(X_rows, index=idxs,
                     columns=['last','mean_1h','std_1h','mean_6h','std_6h','slope'])
    y = pd.Series(y, index=idxs)
    return X, y

# ---------------------------
# 估计全局 nu（MLE）给定标准化残差 u
# ---------------------------
def estimate_nu_mle(u, nu_init=5.0):
    # negative log-likelihood of Student-t for samples u with scale=1 (we already standardized)
    def nll(nu):
        if nu <= 2.01:  # keep >2 for variance existence; guard
            return 1e9
        # using scipy t logpdf
        return -np.sum(t.logpdf(u, df=nu, loc=0, scale=1.0))
    res = minimize(lambda x: nll(x[0]), x0=np.array([nu_init]), bounds=[(2.01, 200.0)])
    if res.success:
        return float(res.x[0])
    else:
        return nu_init

# ---------------------------
# 主流程：训练 μ 模型、训练 logσ 模型、估计 ν、预测区间
# ---------------------------
def train_two_stage(X_train, y_train, X_valid=None, y_valid=None):
    # 1) Model mu (L2 regression)
    params = {'objective':'regression','metric':'l2','verbosity':-1,'num_leaves':31,'learning_rate':0.05}
    dtrain = lgb.Dataset(X_train, label=y_train)
    valid_sets = [dtrain]
    if X_valid is not None:
        dval = lgb.Dataset(X_valid, label=y_valid)
        valid_sets = [dtrain, dval]
    model_mu = lgb.train(params, dtrain, valid_sets=valid_sets, num_boost_round=100, early_stopping_rounds=20, verbose_eval=False)

    # 2) compute residuals on training set
    mu_hat_train = model_mu.predict(X_train)
    resid = y_train.values - mu_hat_train
    eps = 1e-6
    s_target = np.log(np.abs(resid) + eps)  # target for log sigma regression

    # 3) Train log_sigma model
    dtrain_s = lgb.Dataset(X_train, label=s_target)
    params_s = {'objective':'regression','metric':'l2','verbosity':-1,'num_leaves':31,'learning_rate':0.05}
    model_logsigma = lgb.train(params_s, dtrain_s, num_boost_round=100, verbose_eval=False)

    # 4) estimate nu on training normalized residuals using predicted sigma
    sigma_hat_train = np.exp(model_logsigma.predict(X_train))
    u_train = resid / (sigma_hat_train + 1e-9)
    nu_hat = estimate_nu_mle(u_train, nu_init=5.0)
    return model_mu, model_logsigma, nu_hat

def predict_interval(model_mu, model_logsigma, nu, X, alpha=0.05):
    mu = model_mu.predict(X)
    logsigma = model_logsigma.predict(X)
    sigma = np.exp(logsigma)
    lower = t.ppf(alpha/2, df=nu, loc=mu, scale=sigma)
    upper = t.ppf(1-alpha/2, df=nu, loc=mu, scale=sigma)
    return mu, sigma, lower, upper

要点与解释：

我用 log(|resid|+ε) 作为 logσ 的代理回归目标（常用、稳健）。你也可以用 0.5*log(resid^2+ε)；两者等价到常数项。

nu 我在训练集上做 MLE：先用 u=(y-μ)/σ 标准化残差，然后最小化 Student-t 的负对数似然求 ν（标量）。你也可以把 ν 设为超参并用 CV 调参。

预测时直接用 scipy.stats.t.ppf 得到上下界。

这个方案的优点：实现简单、数值稳定、易部署；缺点：ν 为全局或按窗口估计（如果你怀疑 ν 随时间变化，可以把 ν 在分段上估计或用小模型去预测 ν，但复杂度增大）。
