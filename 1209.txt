import numpy as np
import pandas as pd
from scipy import stats
from scipy.fft import rfft, rfftfreq
from scipy.signal import welch, find_peaks
from statsmodels.tsa.stattools import acf, pacf
import tsflex
from tsflex.features import FeatureCollection, MultipleFeatureDescriptors
from tsflex.features.integrations import scipy_stats, statsmodels_ts

# ------------------- 1. 核心特征函数定义 -------------------
def top5_energy_ratio(x: np.ndarray) -> float:
    """前5个FFT峰值能量占比（强周期神器）"""
    N = len(x)
    if N < 64: return 0.0
    yf = np.abs(rfft(x - x.mean()))**2
    peaks, _ = find_peaks(yf, height=np.mean(yf)*2)
    if len(peaks) == 0: return 0.0
    sorted_peaks = peaks[np.argsort(yf[peaks])[::-1]]
    top5 = yf[sorted_peaks[:5]].sum()
    return top5 / yf.sum()

def day_cycle_strength(x: np.ndarray, fs: float = 2.0001) -> float:
    """日周期强度（假设 1min~5min 采样，fs=2.0001 代表 1min）"""
    N = len(x)
    if N < 48: return 0.0
    freqs = rfftfreq(N, 1/fs)
    yf = np.abs(rfft(x - x.mean()))**2
    # 日周期频率 ≈ 1/1440 (1天=1440分钟) 或 1/288 (5min采样)
    target_freq = 1 / 288.0 if fs == 2.0001 else 1 / 1440.0
    idx = np.argmin(np.abs(freqs - target_freq))
    bandwidth = 3
    band_power = yf[max(0, idx-bandwidth):idx+bandwidth+1].sum()
    return band_power / yf.sum()

def kurtosis_outlier_ratio(x: np.ndarray) -> float:
    """峰度 + 3σ 异常点比例组合特征"""
    k = stats.kurtosis(x)
    outlier_ratio = np.mean(np.abs(x - x.mean()) > 3 * x.std())
    return float(k + outlier_ratio * 10)  # 组合成一个强特征

def high_freq_energy_ratio(x: np.ndarray) -> float:
    """高频能量占比（>0.2 归一化频率）"""
    N = len(x)
    if N < 64: return 0.0
    yf = np.abs(rfft(x - x.mean()))**2
    freqs = rfftfreq(N, 1/2.0001)
    high_mask = freqs > 0.2
    return yf[high_mask].sum() / yf.sum() if high_mask.sum() > 0 else 0.0

# ------------------- 2. 构建 tsflex FeatureCollection -------------------
def get_aiops_tsflex_features(window_sec: int = 512):
    """
    返回 AIOps 专用 72 维特征集合
    window_sec: 你的序列长度（如 512 点）
    """
    # 基础统计量（scipy.stats 集成）
    basic_stats = [
        "mean", "std", "var", "skew", "kurtosis",
        "quantile__q=0.9", "quantile__q=0.95", "quantile__q=0.99",
        "range", "iqr"
    ]

    # 自定义函数列表
    custom_funcs = [
        top5_energy_ratio,
        day_cycle_strength,
        high_freq_energy_ratio,
        kurtosis_outlier_ratio,
        lambda x: acf(x, nlags=24, fft=True)[24] if len(x) > 24 else 0.0,  # acf_lag24
        lambda x: acf(x, nlags=168, fft=True)[168] if len(x) > 168 else 0.0,  # acf_lag168
        lambda x: np.mean(np.abs(np.diff(x))),  # mean_abs_change
        lambda x: ((x[:-1] * x[1:]) < 0).sum() / len(x),  # zero_crossing_rate
    ]

    # 构建特征描述器
    feature_descriptors = []

    # 1. 基础统计
    for func in basic_stats:
        feature_descriptors.append((func, "signal__"))

    # 2. 自定义强特征
    custom_names = [
        "top5_energy_ratio", "day_cycle_strength", "high_freq_energy_ratio",
        "kurtosis_outlier_combo", "acf_lag24", "acf_lag168",
        "mean_abs_change", "zero_crossing_rate"
    ]
    for name, func in zip(custom_names, custom_funcs):
        feature_descriptors.append((func, "signal__", name))

    # 3. 更多实用特征（scipy + numpy）
    feature_descriptors.extend([
        ("entropy", "signal__"),
        ("rms", "signal__"),
        ("peak_count", "signal__", lambda x: len(find_peaks(x, height=x.std()*2)[0])),
        ("cid_ce", "signal__", lambda x: np.sqrt(np.sum(np.diff(x)**2))),
        ("spectral_entropy", "signal__", lambda x: stats.entropy(np.abs(rfft(x - x.mean()))**2 + 1e-12)),
    ])

    # 创建 FeatureCollection
    fc = FeatureCollection(
        MultipleFeatureDescriptors(
            functions=feature_descriptors,
            series_names=["signal"],
            windows=window_sec,
            strides=window_sec,  # 不重叠
        )
    )
    return fc

# ------------------- 3. 一键提取函数（复制粘贴就用） -------------------
def extract_aiops_features_tsflex(data: pd.DataFrame, 
                                  window_sec: int = 512) -> pd.DataFrame:
    """
    输入 data: 
        - 列名为指标名（如 "cpu_001", "qps_web")
        - 行为时间戳（等间距）
        - 每列是一个完整序列（长度 >= window_sec）
    输出: pd.DataFrame (n_series, 72)
    """
    fc = get_aiops_tsflex_features(window_sec=window_sec)
    
    # tsflex 支持直接传入 DataFrame（自动按列切窗）
    features_df = fc.calculate(data, return_df=True, n_jobs=-1, show_progress=True)
    
    # 列名处理（去掉多余前缀）
    features_df.columns = [col.replace("signal__", "").replace(f"_{window_sec}", "") 
                          for col in features_df.columns]
    
    print(f"提取完成！共 {features_df.shape[0]} 条序列，{features_df.shape[1]} 维特征")
    return features_df
